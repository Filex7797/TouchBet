def _migrate_legacy_touchbet_db():
    """Move the pre-/app/data SQLite database safely on the first full deploy.

    SQLite backup() is used instead of a raw copy so a leftover WAL file cannot
    silently make balances/bets disappear. Existing persistent DB is never
    overwritten.
    """
    try:
        target = os.path.abspath(DB_NAME)
        if os.path.exists(target) and os.path.getsize(target) > 0:
            return False
        candidates = []
        env_old = (os.getenv("TOUCHBET_LEGACY_DB_PATH", "") or "").strip()
        if env_old:
            candidates.append(env_old)
        candidates.extend([os.path.join(os.getcwd(), "touchbet.db"), "/app/touchbet.db"])
        seen = set()
        for old in candidates:
            old = os.path.abspath(old)
            if old in seen or old == target:
                continue
            seen.add(old)
            if not os.path.isfile(old) or os.path.getsize(old) <= 0:
                continue
            os.makedirs(os.path.dirname(target) or ".", exist_ok=True)
            src = sqlite3.connect(old, timeout=15)
            dst = sqlite3.connect(target, timeout=15)
            try:
                src.backup(dst)
                dst.commit()
            finally:
                dst.close(); src.close()
            return True
    except Exception as exc:
        print(f"TouchBet DB migration warning: {type(exc).__name__}: {exc}")
    return False

_migrate_legacy_touchbet_db()

def local_now() -> datetime:
    """Локальное время TouchBet, не зависящее от timezone операционной системы."""
    return datetime.now(LOCAL_TZ).replace(tzinfo=None)

def now_str() -> str:
    return local_now().strftime("%Y-%m-%d %H:%M:%S")

def esc(value: Any) -> str:
    if value is None:
        return "—"
    return html.escape(str(value))

def money(value: Any) -> str:
    try:
        v = float(value)
        if v.is_integer():
            return str(int(v))
        return f"{v:.2f}"
    except Exception:
        return str(value)

def user_title(user) -> str:
    username = f"@{user.username}" if getattr(user, "username", None) else "—"
    first = getattr(user, "first_name", None) or "—"
    return f"{esc(first)} | {esc(username)} | <code>{user.id}</code>"

def parse_float(text: str) -> Optional[float]:
    try:
        return float(text.replace(",", ".").strip())
    except Exception:
        return None


def begin_immediate(con: sqlite3.Connection) -> None:

def calc_payout(amount: int, odds: float) -> int:
    value = Decimal(str(int(amount))) * Decimal(str(odds))
    return int(value.quantize(Decimal("1"), rounding=ROUND_HALF_UP))


def change_balance(
    cur: sqlite3.Cursor,
    user_id: int,
    delta: int,
    tx_type: str,
    reference_type: str | None = None,
    reference_id: Any = None,
    admin_id: int | None = None,
    earned_delta: int = 0,
) -> tuple[bool, int | None]:
    delta = int(delta)
    earned_delta = int(earned_delta)
    cur.execute(
        """UPDATE users
           SET balance=balance+?, earned_from_bets=earned_from_bets+?
           WHERE id=? AND balance+? >= 0""",
        (delta, earned_delta, int(user_id), delta),
    )
    if cur.rowcount != 1:
        return False, None
    cur.execute("SELECT balance FROM users WHERE id=?", (int(user_id),))
    row = cur.fetchone()
    if not row:
        return False, None
    after = int(row[0])
    before = after - delta
    cur.execute(
        """INSERT INTO transactions
           (user_id, tx_type, amount, balance_before, balance_after, reference_type, reference_id, admin_id, created_at)
           VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
        (int(user_id), tx_type, delta, before, after, reference_type, None if reference_id is None else str(reference_id), admin_id, now_str()),
    )
    return True, after


def match_odds_for_pick(match: sqlite3.Row, pick: str) -> Optional[float]:
    if pick == "1":
        return float(match["odds1"])
    if pick == "2":
        return float(match["odds2"])
    if pick == "X" and match["format"] == "league" and match["oddsx"] is not None:
        return float(match["oddsx"])
    return None


def event_odds_for_pick(event: sqlite3.Row, pick: str) -> Optional[float]:
    if pick == "1":
        return float(event["odds1"])
    if pick == "2":
        return float(event["odds2"])
    return None


def normalize_club_key(value: Any) -> str:
    text = re.sub(r"\s+", " ", str(value or "").strip())
    text = text.casefold().replace("ё", "е")
    text = text.replace("–", "-").replace("—", "-")
    return text


def normalize_club_compact(value: Any) -> str:
    return re.sub(r"[^a-zа-я0-9]+", "", normalize_club_key(value), flags=re.IGNORECASE)


_ROSTER_CACHE: dict[str, Any] = {"mtime": None, "data": None, "error": None, "path": None}


def _parse_registry_time(value: Any) -> Optional[datetime]:
    raw = str(value or "").strip()
    if not raw:
        return None
    try:
        dt = datetime.fromisoformat(raw.replace("Z", "+00:00"))
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        return dt.astimezone(timezone.utc)
    except Exception:
        return None


def _roster_age_seconds(data: dict[str, Any]) -> Optional[int]:
    dt = _parse_registry_time(data.get("generated_at"))
    if not dt:
        return None
    return max(0, int((datetime.now(timezone.utc) - dt).total_seconds()))


def _read_json_cached(path: str) -> tuple[Any, Optional[str]]:
    try:
        stat = os.stat(path)
        mtime = (stat.st_mtime_ns, stat.st_size)
        if _ROSTER_CACHE.get("path") == path and _ROSTER_CACHE.get("mtime") == mtime:
            return _ROSTER_CACHE.get("data"), _ROSTER_CACHE.get("error")
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        _ROSTER_CACHE.update({"path": path, "mtime": mtime, "data": data, "error": None})
        return data, None
    except Exception as e:
        _ROSTER_CACHE.update({"path": path, "mtime": None, "data": None, "error": str(e)})
        return None, str(e)


def _load_external_roster() -> tuple[dict[str, Any] | None, str | None]:
    """Load the live Transfer Markt export.

    In strict mode TouchBet refuses new bets when the live registry is missing,
    broken or stale. This prevents a TM outage from becoming a way to bet on
    your own club after a transfer.
    """
    try:
        if os.path.exists(TOUCHBET_ROSTER_FILE):
            data, read_error = _read_json_cached(TOUCHBET_ROSTER_FILE)
            if read_error:
                return None, f"ошибка чтения {TOUCHBET_ROSTER_FILE}: {read_error}"
            players = data.get("players") if isinstance(data, dict) else None
            if not isinstance(players, dict):
                return None, "в touchbet_roster.json нет секции players"
            age = _roster_age_seconds(data)
            # V55+ exports have generated_at and are refreshed every minute.
            if int(data.get("schema") or 0) >= 2:
                if age is None:
                    return None, "в реестре TM нет корректного generated_at"
                if age > TOUCHBET_ROSTER_MAX_AGE_SECONDS:
                    return None, f"реестр TM устарел ({age} сек.)"
            return data, None

        # Backward compatibility when both bots still run in one directory.
        # This fallback has no freshness metadata, so live V55 export is preferred.
        if os.path.exists(TOUCHBET_USERS_FALLBACK_FILE):
            users, read_error = _read_json_cached(TOUCHBET_USERS_FALLBACK_FILE)
            if read_error:
                return None, read_error
            if not isinstance(users, dict):
                return None, "users.json имеет неверный формат"
            clubs = sorted({str(u.get("club")) for u in users.values() if isinstance(u, dict) and u.get("club")})
            return {"version": 0, "schema": 0, "clubs": clubs, "nations": [], "aliases": {}, "players": users}, None
        return None, f"файл реестра не найден: {TOUCHBET_ROSTER_FILE}"
    except Exception as e:
        logger.warning("Ошибка чтения реестра клубов: %s", e)
        return None, str(e)


def _registry_alias_index(roster: dict[str, Any]) -> dict[str, str]:
    index: dict[str, str] = {}
    all_teams = list(roster.get("clubs", []) or []) + list(roster.get("nations", []) or [])
    aliases = roster.get("aliases") or {}
    for canonical in all_teams:
        canonical = str(canonical).strip()
        if not canonical:
            continue
        for variant in [canonical] + list(aliases.get(canonical, []) or []):
            v = str(variant or "").strip()
            if not v:
                continue
            index[normalize_club_key(v)] = canonical
            index[normalize_club_compact(v)] = canonical
    return index


def canonical_registry_club(name: str) -> str:
    """Resolve Russian/English spellings to one TM team name."""
    value = re.sub(r"\s+", " ", str(name or "").strip())
    roster, _ = _load_external_roster()
    if not roster:
        return value
    index = _registry_alias_index(roster)
    for key in (normalize_club_key(value), normalize_club_compact(value)):
        if key in index:
            return str(index[key])

    # Conservative fuzzy matching only within the same alphabet. It helps with
    # small typos but won't magically equate unrelated clubs.
    candidates = sorted(set(index.values()))
    needle = normalize_club_key(value)
    if needle:
        same_script = []
        latin = bool(re.search(r"[a-z]", needle)) and not bool(re.search(r"[а-я]", needle))
        cyr = bool(re.search(r"[а-я]", needle)) and not bool(re.search(r"[a-z]", needle))
        for team in candidates:
            variants = [team] + list((roster.get("aliases") or {}).get(team, []) or [])
            for variant in variants:
                n = normalize_club_key(variant)
                if latin and re.search(r"[а-я]", n):
                    continue
                if cyr and re.search(r"[a-z]", n):
                    continue
                ratio = difflib.SequenceMatcher(None, needle, n).ratio()
                if ratio >= 0.90:
                    same_script.append((ratio, team))
        same_script.sort(reverse=True)
        if same_script and (len(same_script) == 1 or same_script[0][0] - same_script[1][0] >= 0.04):
            return str(same_script[0][1])
    return value


def get_player_protected_teams(user_id: int) -> tuple[list[str], Optional[str]]:
    roster, error = _load_external_roster()
    if error:
        return [], error
    record = (roster.get("players") or {}).get(str(user_id))
    if not isinstance(record, dict):
        return [], None
    if not bool(record.get("career_active", True)):
        return [], None
    values = record.get("protected_teams") if isinstance(record.get("protected_teams"), list) else None
    if values is None:
        values = [record.get("club"), record.get("club_owner"), record.get("nation"), record.get("nation_owner")]
    result = []
    for value in values:
        if not value:
            continue
        canonical = canonical_registry_club(str(value))
        if canonical and canonical not in result:
            result.append(canonical)
    return result, None


def get_player_current_club(user_id: int) -> tuple[Optional[str], Optional[str]]:
    """Compatibility helper used by older TouchBet code."""
    roster, error = _load_external_roster()
    if error:
        return None, error
    record = (roster.get("players") or {}).get(str(user_id))
    if not isinstance(record, dict) or not bool(record.get("career_active", True)):
        return None, None
    club = record.get("club") or record.get("club_owner")
    if not club:
        return None, None
    return canonical_registry_club(str(club)), None


def check_player_club_for_teams(user_id: int, *team_names: str) -> tuple[bool, str]:
    protected, error = get_player_protected_teams(user_id)
    if error:
        if STRICT_CLUB_CHECK:
            return False, (
                "⚠️ Сейчас не удалось получить свежие данные из Transfer Markt. "
                "Ставка временно недоступна, чтобы нельзя было обойти запрет на ставки на свои команды."
            )
        logger.warning("Проверка команд пропущена для %s: %s", user_id, error)
        return True, ""
    if not protected:
        return True, ""
    protected_keys = {normalize_club_key(canonical_registry_club(x)): x for x in protected}
    for team in team_names:
        if not team:
            continue
        canonical = canonical_registry_club(str(team))
        key = normalize_club_key(canonical)
        if key in protected_keys:
            own = protected_keys[key]
            return False, (
                f"⛔ Ставки на матчи/исходы с участием вашей команды <b>{esc(own)}</b> запрещены. "
                "Состав и владение синхронизируются автоматически из Transfer Markt."
            )
    return True, ""


def touchbet_registry_health() -> tuple[bool, str, Optional[int]]:
    roster, error = _load_external_roster()
    if error or not roster:
        return False, error or "нет данных", None
    age = _roster_age_seconds(roster)
    players = len(roster.get("players") or {})
    clubs = len(roster.get("clubs") or [])
    if age is None:
        return True, f"legacy sync • {players} игроков • {clubs} клубов", None
    return True, f"{players} игроков • {clubs} клубов • возраст {age} сек.", age


def match_time_has_started(value: str | None) -> bool:
    if not value:
        return False
    mt = parse_db_datetime(value)
    if not mt:
        logger.error("Invalid match_time value while checking event bets: %r", value)
        return True
    return local_now() >= mt


def parse_purchase_payload(payload: str) -> Optional[dict[str, Any]]:
    parts = str(payload or "").split(":")
    if len(parts) != 4 or parts[0] not in {"premium", "coins"}:
        return None
    try:
        return {"kind": parts[0], "user_id": int(parts[1]), "stars": int(parts[2]), "nonce": parts[3]}
    except (TypeError, ValueError):
        return None


def validate_purchase_payload(payload: str, user_id: int, total_amount: int, currency: str) -> tuple[Optional[dict[str, Any]], Optional[str]]:
    data = parse_purchase_payload(payload)
    if not data:
        return None, "неверный payload"
    if currency != "XTR":
        return None, "неверная валюта"
    if data["user_id"] != int(user_id):
        return None, "payload принадлежит другому пользователю"
    if int(total_amount) != data["stars"]:
        return None, "сумма платежа не совпадает с payload"
    if data["kind"] == "premium" and data["stars"] != PREMIUM_PRICE_STARS:
        return None, "неверная цена premium"
    if data["kind"] == "coins" and data["stars"] not in STAR_PACKAGES:
        return None, "неизвестный пакет коинов"
    return data, None


def parse_match_datetime(text: str) -> Optional[str]:
    raw = (text or "").strip()
    if raw in {"-", "нет", "Нет", "no", "NO"}:
        return None
    for fmt in ("%d.%m.%Y %H:%M", "%d.%m.%y %H:%M", "%Y-%m-%d %H:%M", "%Y-%m-%d %H:%M:%S"):
        try:
            return datetime.strptime(raw, fmt).strftime("%Y-%m-%d %H:%M:%S")
        except Exception:
            pass
    return "INVALID"

def match_is_closed_for_bets(m: sqlite3.Row) -> bool:
    """Single source of truth: at the exact scheduled minute betting is closed."""
    if not m:
        return True
    try:
        if m["status"] != "active":
            return True
    except Exception:
        return True
    try:
        mt = m["match_time"]
    except Exception:
        mt = None
    if not mt:
        return False
    parsed = parse_db_datetime(str(mt))
    if not parsed:
        # Bad stored time must fail closed rather than accidentally accept bets.
        logger.error("Invalid match_time on active match id=%s value=%r", (m["id"] if "id" in m.keys() else "?"), mt)
        return True
    return local_now() >= parsed

async def safe_answer_callback(callback: CallbackQuery, text: str = "", show_alert: bool = False):
    try:
        await callback.answer(text, show_alert=show_alert)
    except TelegramBadRequest:
        pass

async def safe_edit_or_answer(target: Message | CallbackQuery, text: str, reply_markup=None):
    if isinstance(target, CallbackQuery):
        await safe_answer_callback(target)
        try:
            await target.message.edit_text(text, reply_markup=reply_markup)
            return
        except TelegramBadRequest:
            pass
        await target.message.answer(text, reply_markup=reply_markup)
        return
    await target.answer(text, reply_markup=reply_markup)

async def deny_private_only(message: Message) -> bool:

def db() -> sqlite3.Connection:
    con = sqlite3.connect(DB_NAME, timeout=15)
    con.row_factory = sqlite3.Row
    con.execute("PRAGMA foreign_keys=ON")
    con.execute("PRAGMA busy_timeout=15000")
    return con

def table_columns(cur: sqlite3.Cursor, table: str) -> set[str]:
    cur.execute(f"PRAGMA table_info({table})")
    return {row[1] for row in cur.fetchall()}

def add_column(cur: sqlite3.Cursor, table: str, name: str, definition: str):
    if name not in table_columns(cur, table):
        cur.execute(f"ALTER TABLE {table} ADD COLUMN {name} {definition}")

def register_user_from_tg(user) -> sqlite3.Row:
    first_name = getattr(user, "first_name", None) or ""
    username = getattr(user, "username", None)
    uid = user.id
    with db() as con:
        cur = con.cursor()
        cur.execute("SELECT * FROM users WHERE id=?", (uid,))
        row = cur.fetchone()
        if not row:
            cur.execute(
                """INSERT INTO users (id, first_name, username, balance, banned, registered_at, created_at, earned_from_bets)
                   VALUES (?, ?, ?, ?, 0, ?, ?, 0)""",
                (uid, first_name, username, START_BALANCE, now_str(), now_str())
            )
        else:
            cur.execute(
                "UPDATE users SET first_name=?, username=?, registered_at=COALESCE(registered_at, ?), created_at=COALESCE(created_at, ?) WHERE id=?",
                (first_name, username, now_str(), now_str(), uid)
            )
        con.commit()
        cur.execute("SELECT * FROM users WHERE id=?", (uid,))
        return cur.fetchone()

def get_user(uid: int) -> Optional[sqlite3.Row]:
    return refresh_premium_status(uid)

def find_user(identifier: str) -> Optional[sqlite3.Row]:
    identifier = identifier.strip()
    with db() as con:
        cur = con.cursor()
        if identifier.startswith("@"):
            cur.execute("SELECT * FROM users WHERE lower(username)=lower(?)", (identifier[1:],))
        elif identifier.isdigit():
            cur.execute("SELECT * FROM users WHERE id=?", (int(identifier),))
        else:
            cur.execute("SELECT * FROM users WHERE lower(username)=lower(?)", (identifier,))
        return cur.fetchone()


def parse_db_datetime(value: str | None) -> Optional[datetime]:
    if not value:
        return None
    for fmt in ("%Y-%m-%d %H:%M:%S", "%d.%m.%Y %H:%M", "%Y-%m-%d %H:%M"):
        try:
            return datetime.strptime(value, fmt)
        except Exception:
            pass
    return None

def refresh_premium_status(uid: int) -> Optional[sqlite3.Row]:
    """Премиум действует 30 дней. Старые записи без premium_until аккуратно получают срок."""
    with db() as con:
        cur = con.cursor()
        cur.execute("SELECT * FROM users WHERE id=?", (uid,))
        row = cur.fetchone()
        if not row:
            return None
        try:
            if row["premium"]:
                until = row["premium_until"] if "premium_until" in row.keys() else None
                until_dt = parse_db_datetime(until)
                if not until_dt:
                    since_dt = parse_db_datetime(row["premium_since"] if "premium_since" in row.keys() else None) or local_now()
                    until_dt = since_dt + timedelta(days=PREMIUM_DAYS)
                    cur.execute("UPDATE users SET premium_until=? WHERE id=?", (until_dt.strftime("%Y-%m-%d %H:%M:%S"), uid))
                    con.commit()
                if local_now() >= until_dt:
                    cur.execute("UPDATE users SET premium=0, premium_since=NULL, premium_until=NULL WHERE id=?", (uid,))
                    con.commit()
        except Exception:
            pass
        cur.execute("SELECT * FROM users WHERE id=?", (uid,))
        return cur.fetchone()

def is_premium_active(uid: int) -> bool:
    row = refresh_premium_status(uid)
    return bool(row and row["premium"])

def is_banned(uid: int) -> tuple[bool, str]:
    user = get_user(uid)
    if not user:
        return False, ""
    if not user["banned"]:
        return False, ""
    until = user["banned_until"]
    reason = user["ban_reason"] or "Без причины"
    if until == "perm":
        return True, f"⛔ Вы забанены навсегда.\nПричина: <b>{esc(reason)}</b>"
    try:
        until_dt = datetime.strptime(until, "%Y-%m-%d %H:%M:%S")
        if local_now() >= until_dt:
            with db() as con:
                con.execute("UPDATE users SET banned=0, banned_until=NULL, ban_reason=NULL WHERE id=?", (uid,))
                con.commit()
            return False, ""
        return True, f"⛔ Вы забанены до <b>{esc(until)}</b>.\nПричина: <b>{esc(reason)}</b>"
    except Exception:
        return True, f"⛔ Вы забанены.\nПричина: <b>{esc(reason)}</b>"

async def guard_user_message(message: Message) -> bool:
    register_user_from_tg(message.from_user)
    banned, text = is_banned(message.from_user.id)
    if banned:
        await message.answer(text)
        return False
    return True

async def guard_user_callback(callback: CallbackQuery) -> bool:
    register_user_from_tg(callback.from_user)
    banned, text = is_banned(callback.from_user.id)
    if banned:
        await safe_answer_callback(callback, "Вы забанены", True)
        try:
            await callback.message.answer(text)
        except Exception:
            pass
        return False
    return True

def ensure_league(name: str, url: str | None = None):
    name = (name or "").strip()
    if not name:
        return
    with db() as con:
        cur = con.cursor()
        cur.execute("SELECT id FROM leagues WHERE lower(name)=lower(?)", (name,))
        existing = cur.fetchone()
        if existing:
            cur.execute("UPDATE leagues SET url=COALESCE(NULLIF(?, ''), url) WHERE id=?", (url or "", existing["id"]))
        else:
            cur.execute("INSERT OR IGNORE INTO leagues (name, url, created_at) VALUES (?, ?, ?)", (name, url, now_str()))
        con.commit()

def event_result_by_score(ev: sqlite3.Row, g1: int, g2: int) -> str:
    line = float(ev["line"] or 0)
    kind = ev["kind"]
    if kind == "total":
        return "1" if (g1 + g2) > line else "2"
    if kind == "it1":
        return "1" if g1 > line else "2"
    if kind == "it2":
        return "1" if g2 > line else "2"
    if kind == "handicap1":
        return "1" if (g1 + line) > g2 else "2"
    if kind == "handicap2":
        return "1" if (g2 + line) > g1 else "2"
    return "2"

def get_user_rank_by_earned(uid: int) -> Optional[int]:
    with db() as con:
        cur = con.cursor()
        cur.execute("""
            SELECT id, RANK() OVER (ORDER BY earned_from_bets DESC, balance DESC) AS rank_num
            FROM users
        """)
        for row in cur.fetchall():
            if row["id"] == uid:
                return row["rank_num"]
    return None

def build_history(uid: int) -> str:

    with db() as con:
        cur = con.cursor()
        cur.execute("""
            SELECT b.*, m.team1, m.team2, m.score, m.result
            FROM bets b
            LEFT JOIN matches m ON m.id=b.match_id
            WHERE b.user_id=?
            ORDER BY b.id DESC LIMIT 10
        """, (uid,))
        bets = cur.fetchall()
        cur.execute("""
            SELECT wb.*, t.league_name
            FROM winner_bets wb
            LEFT JOIN tournaments t ON t.id=wb.tournament_id
            WHERE wb.user_id=?
            ORDER BY wb.id DESC LIMIT 10
        """, (uid,))
        wbets = cur.fetchall()
        cur.execute("""
            SELECT eb.*, ev.title, ev.option1, ev.option2, m.team1, m.team2
            FROM event_bets eb
            LEFT JOIN match_events ev ON ev.id=eb.event_id
            LEFT JOIN matches m ON m.id=eb.match_id
            WHERE eb.user_id=?
            ORDER BY eb.id DESC LIMIT 10
        """, (uid,))
        ebets = cur.fetchall()
    if not bets and not wbets and not ebets:
        return "<b>📜 История ставок</b>\n\nПока ставок нет."
    if bets:
        parts.append("\n<b>⚽ Матчи:</b>")
        for b in bets:

def build_top(limit: int = 10, by_balance: bool = False) -> str:
    order = "balance DESC" if by_balance else "earned_from_bets DESC, balance DESC"
    title = "<b>💰 Топ игроков по балансу</b>" if by_balance else "<b>🏅 Топ игроков по заработку со ставок</b>"
    with db() as con:
        cur = con.cursor()
        cur.execute(f"SELECT * FROM users ORDER BY {order} LIMIT ?", (limit,))
        rows = cur.fetchall()
    if not rows:
        return title + "\n\nПока игроков нет."
    text = [title, ""]
    for i, row in enumerate(rows, 1):
        name = row["first_name"] or row["username"] or str(row["id"])
        if by_balance:
            value = f"{money(row['balance'])} коинов"
        else:
            value = f"{money(row['earned_from_bets'])} коинов"
        text.append(f"{i}. <b>{esc(name)}</b> — {value}")
    return "\n".join(text)

def seconds_until_daily(last_time: str | None) -> int:
    if not last_time:
        return 0
    try:
        last_dt = datetime.strptime(last_time, "%Y-%m-%d %H:%M:%S")
        next_dt = last_dt + timedelta(hours=24)
        return max(0, int((next_dt - local_now()).total_seconds()))
    except Exception:
        return 0

def format_left(seconds: int) -> str:
    if seconds <= 0:
        return "0 мин"
    hours = seconds // 3600
    minutes = (seconds % 3600) // 60
    if hours > 0:
        return f"{hours}ч {minutes}м"
    return f"{minutes}м"

async def give_daily_bonus_to_user(user_id: int) -> tuple[bool, str]:
    refresh_premium_status(user_id)
    with db() as con:
        begin_immediate(con)
        cur = con.cursor()
        cur.execute("SELECT * FROM users WHERE id=?", (user_id,))
        user = cur.fetchone()
        if not user:
            con.rollback()
            return False, "❌ Пользователь не найден."
        left = seconds_until_daily(user["last_daily_bonus_at"])
        if left > 0:
            con.rollback()
            return False, f"⏳ Ежедневный бонус уже получен. Следующий будет доступен через <b>{format_left(left)}</b>."
        amount = PREMIUM_DAILY_BONUS_COINS if user["premium"] else DAILY_BONUS_COINS
        ok, _ = change_balance(cur, user_id, amount, "daily_bonus")
        if not ok:
            con.rollback()
            return False, "❌ Не удалось начислить бонус. Попробуйте позже."
        cur.execute("UPDATE users SET last_daily_bonus_at=? WHERE id=?", (now_str(), user_id))
        con.commit()
    return True, f"✅ Ежедневный бонус получен!\n\n💰 Начислено: <b>{amount}</b> коинов."

def can_cancel_by_match_time(match_time: str | None) -> bool:
    mt = parse_db_datetime(match_time)
    if not mt:
        return False
    return local_now() < mt

@router.callback_query(lambda c: c.data and c.data.startswith("cancelbet:"))
async def cb_cancel_bet_do(callback: CallbackQuery):
    if not await guard_user_callback(callback):
        return
    if not is_premium_active(callback.from_user.id):
        await safe_answer_callback(callback, "Нужен премиум", True)
        return
    parts = callback.data.split(":")
    if len(parts) != 3 or parts[1] not in {"b", "e"} or not parts[2].isdigit():
        await safe_answer_callback(callback, "Ошибка кнопки", True)
        return
    kind, bid = parts[1], int(parts[2])
    table = "bets" if kind == "b" else "event_bets"
    with db() as con:
        begin_immediate(con)
        cur = con.cursor()
        if table == "bets":
            cur.execute("""
                SELECT b.*, m.team1, m.team2, m.match_time, m.status AS match_status
                FROM bets b JOIN matches m ON m.id=b.match_id
                WHERE b.id=? AND b.user_id=?
            """, (bid, callback.from_user.id))
        else:
            cur.execute("""
                SELECT eb.*, ev.title, m.team1, m.team2, m.match_time, m.status AS match_status
                FROM event_bets eb
                JOIN match_events ev ON ev.id=eb.event_id
                JOIN matches m ON m.id=eb.match_id
                WHERE eb.id=? AND eb.user_id=?
            """, (bid, callback.from_user.id))
        bet = cur.fetchone()
        if not bet or bet["status"] != "pending":
            con.rollback()
            await safe_edit_or_answer(callback, "❌ Ставка не найдена или уже рассчитана.", profile_kb())
            return
        if bet["match_status"] != "active" or not can_cancel_by_match_time(bet["match_time"]):
            con.rollback()
            await safe_edit_or_answer(callback, "⛔ Эту ставку уже нельзя отменить: матч начался, завершён или время матча не указано.", profile_kb())
            return
        cur.execute(f"UPDATE {table} SET status='refunded', payout=? WHERE id=? AND status='pending'", (bet["amount"], bid))
        if cur.rowcount != 1:
            con.rollback()
            await safe_edit_or_answer(callback, "❌ Ставка уже была обработана.", profile_kb())
            return
        ok, _ = change_balance(
            cur, callback.from_user.id, int(bet["amount"]), "premium_refund",
            "bet" if kind == "b" else "event_bet", bid,
        )
        if not ok:
            con.rollback()
            await safe_edit_or_answer(callback, "❌ Не удалось вернуть коины. Обратитесь в поддержку.", profile_kb())
            return
        con.commit()
    await safe_edit_or_answer(callback, f"✅ Ставка отменена.\n\n↩️ Возвращено: <b>{bet['amount']}</b> коинов", profile_kb())
    await log_admin(f"↩️ <b>Премиум-отмена ставки</b>\nИгрок: {user_title(callback.from_user)}\nТип: <b>{'матч' if kind == 'b' else 'событие'}</b>\nID ставки: <code>{bid}</code>\nВозврат: <b>{bet['amount']}</b>")

@router.message(PromoActivateFSM.code)
async def st_promo_code(message: Message, state: FSMContext):
    if await deny_private_only(message): return
    if not await guard_user_message(message): return
    if not await check_channel_subscription(message.from_user.id):
        await message.answer(subscription_text(), reply_markup=back_menu_kb()); await state.clear(); return
    code = (message.text or "").strip()
    with db() as con:
        begin_immediate(con)
        cur = con.cursor()
        cur.execute("SELECT * FROM promo_codes WHERE lower(code)=lower(?) AND active=1", (code,))
        promo = cur.fetchone()
        if not promo:
            con.rollback(); await message.answer("❌ Промокод не найден или выключен."); await state.clear(); return
        if promo["expires_at"] and promo["expires_at"] != "perm":
            try:
                if local_now() >= datetime.strptime(promo["expires_at"], "%Y-%m-%d %H:%M:%S"):
                    cur.execute("UPDATE promo_codes SET active=0 WHERE id=?", (promo["id"],)); con.commit()
                    await message.answer("❌ Срок действия промокода истёк."); await state.clear(); return
            except Exception as e:
                logger.warning("Некорректный срок промокода %s: %s", promo["id"], e)
        if promo["used_count"] >= promo["max_activations"]:
            con.rollback(); await message.answer("❌ Лимит активаций промокода закончился."); await state.clear(); return
        cur.execute("SELECT 1 FROM promo_activations WHERE promo_id=? AND user_id=?", (promo["id"], message.from_user.id))
        if cur.fetchone():
            con.rollback(); await message.answer("⚠️ Вы уже активировали этот промокод."); await state.clear(); return
        try:
            cur.execute("INSERT INTO promo_activations (promo_id, user_id, activated_at) VALUES (?, ?, ?)", (promo["id"], message.from_user.id, now_str()))
        except sqlite3.IntegrityError:
            con.rollback(); await message.answer("⚠️ Вы уже активировали этот промокод."); await state.clear(); return
        cur.execute("UPDATE promo_codes SET used_count=used_count+1 WHERE id=? AND used_count < max_activations", (promo["id"],))
        if cur.rowcount != 1:
            con.rollback(); await message.answer("❌ Лимит активаций промокода закончился."); await state.clear(); return
        ok, _ = change_balance(cur, message.from_user.id, int(promo["reward"]), "promo", "promo", promo["id"])
        if not ok:
            con.rollback(); await message.answer("❌ Не удалось начислить промокод. Обратитесь в поддержку."); await state.clear(); return
        con.commit()
    await state.clear()
    await message.answer(f"✅ Промокод активирован!\n\n💰 Начислено: <b>{promo['reward']}</b> коинов", reply_markup=back_menu_kb())
    await log_admin(f"🎁 <b>Промокод активирован</b>\nКод: <code>{esc(code)}</code>\nИгрок: {user_title(message.from_user)}\nКоины: <b>{promo['reward']}</b>")

async def settle_events_for_match(cur: sqlite3.Cursor, match_id: int, g1: int, g2: int) -> list[sqlite3.Row]:
    cur.execute("SELECT * FROM matches WHERE id=?", (match_id,))
    match_row = cur.fetchone()
    cur.execute("SELECT * FROM match_events WHERE match_id=? AND status='active'", (match_id,))
    events = cur.fetchall()
    for ev in events:
        result = event_result_by_score(ev, g1, g2)
        cur.execute("UPDATE match_events SET status='finished', result_option=? WHERE id=?", (result, ev["id"]))
        cur.execute("SELECT * FROM event_bets WHERE event_id=? AND status='pending'", (ev["id"],))
        bets = cur.fetchall()
        for b in bets:
            # Re-check current TM membership at settlement. If a player joined
            # one of the teams after placing the bet, the bet is void/refunded.
            if match_row:
                allowed, _ = check_player_club_for_teams(b["user_id"], match_row["team1"], match_row["team2"])
            else:
                allowed = True
            if not allowed:
                change_balance(cur, b["user_id"], int(b["amount"]), "tm_own_team_refund", "event_bet", b["id"])
                cur.execute("UPDATE event_bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
                continue
            if b["pick"] == result:
                payout = calc_payout(b["amount"], b["odds"])
                profit = payout - int(b["amount"])
                change_balance(cur, b["user_id"], payout, "event_bet_win", "event_bet", b["id"], earned_delta=profit)
                cur.execute("UPDATE event_bets SET status='won', payout=? WHERE id=? AND status='pending'", (payout, b["id"]))
            else:
                cur.execute("UPDATE event_bets SET status='lost', payout=0 WHERE id=? AND status='pending'", (b["id"],))
    return events

            con.rollback(); await message.answer("⛔ Нельзя ставить на победу своего текущего клуба."); await state.clear(); return
        odds, team_name = float(team["odds"]), team["name"]
        try:
            cur.execute("""INSERT INTO winner_bets
                (user_id, tournament_id, team_id, team_name, amount, odds, status, tournament_version, created_at)
                VALUES (?, ?, ?, ?, ?, ?, 'pending', ?, ?)""",
                (message.from_user.id, tid, team_id, team_name, amount, odds, int(t["version"] or 1), now_str()))
            bet_id = cur.lastrowid
        except sqlite3.IntegrityError:
            con.rollback(); await message.answer("⚠️ Такая ставка уже существует."); await state.clear(); return
        ok, _ = change_balance(cur, message.from_user.id, -amount, "tournament_bet_stake", "winner_bet", bet_id)
        if not ok:
            con.rollback(); await message.answer("❌ Недостаточно коинов."); return
        con.commit()
    await state.clear()
    await message.answer(
        "✅ <b>Ставка на победителя принята</b>\n\n"
        f"🏆 Турнир: <b>{esc(t['league_name'])}</b>\n"
        f"🎯 Команда: <b>{esc(team_name)}</b>\n"
        f"💰 Сумма: <b>{amount}</b> коинов\n"
        f"📊 Кэф: <b>{odds}</b>\n"
        f"🏆 Возможная выплата: <b>{calc_payout(amount, odds)}</b> коинов",
        reply_markup=back_menu_kb()
    )
    await log_admin(f"🏆 <b>Новая ставка на победителя</b>\nИгрок: {user_title(message.from_user)}\nТурнир: <b>{esc(t['league_name'])}</b>\nКоманда: <b>{esc(team_name)}</b>\nСумма: <b>{amount}</b> | Кэф: <b>{odds}</b>")

        await safe_answer_callback(callback, "Нет доступа", True); return

        await safe_edit_or_answer(callback, build_match_preview(await state.get_data()), confirm_cancel_kb("match:create_ok", "match:create_cancel"))
        return
    if action == "done":
        if not selected:
            await state.update_data(events=[])
            await state.set_state(MatchCreate.confirm)
            await safe_edit_or_answer(callback, build_match_preview(await state.get_data()), confirm_cancel_kb("match:create_ok", "match:create_cancel"))
            return

        await state.set_state(MatchCreate.event_config)
        await safe_edit_or_answer(callback, get_next_event_prompt(await state.get_data()))
        return
    if action in EVENT_TYPES:
        if action in selected:
            selected.remove(action)
        else:
            selected.append(action)
        await state.update_data(selected_events=selected)
        await safe_edit_or_answer(callback, "📌 Выберите события для матча. Можно выбрать несколько или все:", event_select_kb(selected))
        return
    await safe_answer_callback(callback, "Неизвестное действие", True)

@router.message(MatchCreate.event_config)
async def st_event_config(message: Message, state: FSMContext):
    if not is_admin_user(message.from_user.id): return
    if message.text == "/cancel":
        await state.clear(); await message.answer("❌ Создание матча отменено.", reply_markup=admin_kb()); return
    data = await state.get_data()
    selected = data.get("selected_events", [])
    idx = int(data.get("event_index", 0))
    if idx >= len(selected):
        await state.set_state(MatchCreate.confirm)
        await message.answer(build_match_preview(data), reply_markup=confirm_cancel_kb("match:create_ok", "match:create_cancel"))
        return
    parts = parse_pipe_parts(message.text)
    if len(parts) != 3:
        await message.answer("❌ Формат: <code>линия | кэф 1 | кэф 2</code>"); return
    line = parse_float(parts[0]); odd1 = parse_float(parts[1]); odd2 = parse_float(parts[2])
    if line is None or not odd1 or not odd2 or odd1 <= 0 or odd2 <= 0:
        await message.answer("❌ Проверьте линию и коэффициенты."); return
    kind = selected[idx]
    payload = create_event_payload(kind, line, odd1, odd2, data.get("team1", "Команда 1"), data.get("team2", "Команда 2"))
    ev = {"kind": kind, "line": line, "odds1": odd1, "odds2": odd2, **payload}
    events = list(data.get("events", []))
    events.append(ev)
    idx += 1
    await state.update_data(events=events, event_index=idx)
    if idx < len(selected):
        await message.answer(get_next_event_prompt(await state.get_data()))
    else:
        await state.set_state(MatchCreate.confirm)
        await message.answer(build_match_preview(await state.get_data()), reply_markup=confirm_cancel_kb("match:create_ok", "match:create_cancel"))

        await state.clear()
        return
    ensure_league(data["league_name"], data.get("league_url"))
    with db() as con:
        cur = con.cursor()
        cur.execute("""INSERT INTO matches
            (league_name, league_url, team1, team2, format, odds1, oddsx, odds2, status, match_time, created_at)

                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'active', ?)""",
                (mid, ev["kind"], ev["title"], ev["option1"], ev["option2"], ev["odds1"], ev["odds2"], ev.get("line"), ev.get("team_side"), now_str()))

    await safe_edit_or_answer(callback, f"✅ Матч #{mid} создан.\n📌 Событий: <b>{len(data.get('events', []))}</b>", admin_kb())

@router.message(MatchFinish.score)
async def st_finish_score(message: Message, state: FSMContext):
    if not is_admin_user(message.from_user.id): return
    if ":" not in (message.text or ""):
        await message.answer("❌ Введите счёт в формате 2:1"); return
    try:
        g1, g2 = [int(x.strip()) for x in message.text.split(":", 1)]
        if g1 < 0 or g2 < 0: raise ValueError
    except Exception:
        await message.answer("❌ Счёт должен быть неотрицательными числами, например 2:1"); return
    data = await state.get_data(); mid = int(data["match_id"])
    result = "1" if g1 > g2 else "2" if g2 > g1 else "X"
    with db() as con:
        begin_immediate(con); cur = con.cursor()
        cur.execute("SELECT * FROM matches WHERE id=? AND status='active'", (mid,)); m = cur.fetchone()
        if not m:
            con.rollback(); await message.answer("❌ Матч не найден или уже завершён."); await state.clear(); return
        sync_ok, sync_detail, _ = touchbet_registry_health()
        if STRICT_CLUB_CHECK and not sync_ok:
            con.rollback(); await message.answer(f"⚠️ Матч пока не рассчитан: нет свежей синхронизации с Transfer Markt.\n\n{esc(sync_detail)}\n\nПовторите завершение после восстановления синхронизации."); return
        if result == "X" and m["format"] == "playoff":
            con.rollback(); await message.answer("❌ В плей-офф не может быть ничьей. Введите финальный счёт с победителем."); return
        cur.execute("UPDATE matches SET status='finished', result=?, score=? WHERE id=? AND status='active'", (result, message.text.strip(), mid))
        if cur.rowcount != 1:
            con.rollback(); await message.answer("❌ Матч уже был завершён другим администратором."); await state.clear(); return
        cur.execute("SELECT * FROM bets WHERE match_id=? AND status='pending'", (mid,)); bets = cur.fetchall()
        tm_refunded_bet_ids = set()
        for b in bets:
            allowed_now, _ = check_player_club_for_teams(b["user_id"], m["team1"], m["team2"])
            if not allowed_now:
                change_balance(cur, b["user_id"], int(b["amount"]), "tm_own_team_refund", "bet", b["id"])
                cur.execute("UPDATE bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
                tm_refunded_bet_ids.add(int(b["id"]))
                continue
            if b["pick"] == result:
                payout = calc_payout(b["amount"], b["odds"]); profit = payout - int(b["amount"])
                change_balance(cur, b["user_id"], payout, "bet_win", "bet", b["id"], earned_delta=profit)
                cur.execute("UPDATE bets SET status='won', payout=? WHERE id=? AND status='pending'", (payout, b["id"]))
            else:
                cur.execute("UPDATE bets SET status='lost', payout=0 WHERE id=? AND status='pending'", (b["id"],))
        cur.execute("""SELECT eb.id FROM event_bets eb LEFT JOIN match_events ev ON ev.id=eb.event_id
                       WHERE eb.match_id=? AND ev.status='active' AND eb.status='pending'""", (mid,))
        pending_event_bet_ids = [row[0] for row in cur.fetchall()]
        await settle_events_for_match(cur, mid, g1, g2)
        if pending_event_bet_ids:
            placeholders = ",".join("?" for _ in pending_event_bet_ids)
            cur.execute(f"""SELECT eb.*, ev.title, ev.option1, ev.option2 FROM event_bets eb
                            LEFT JOIN match_events ev ON ev.id=eb.event_id WHERE eb.id IN ({placeholders}) ORDER BY eb.id DESC""", pending_event_bet_ids)
            event_bets_done = cur.fetchall()
        else:
            event_bets_done = []
        con.commit()
    await settle_miniapp_slips_for_match(mid, g1, g2)
    for b in bets:
        try:
            if int(b["id"]) in tm_refunded_bet_ids:
                await bot.send_message(b["user_id"], f"↩️ Ставка аннулирована и возвращена.\n\nВы сейчас состоите в одной из команд матча <b>{esc(m['team1'])} — {esc(m['team2'])}</b>.\nВозврат: <b>{b['amount']}</b> коинов")
            elif b["pick"] == result:
                payout = calc_payout(b["amount"], b["odds"])
                await bot.send_message(b["user_id"], f"✅ Ваша ставка выиграла!\n\nМатч: <b>{esc(m['team1'])} — {esc(m['team2'])}</b>\nСчёт: <b>{esc(message.text)}</b>\nВыигрыш: <b>{payout}</b> коинов")
            else:
                await bot.send_message(b["user_id"], f"❌ Ваша ставка проиграла.\n\nМатч: <b>{esc(m['team1'])} — {esc(m['team2'])}</b>\nСчёт: <b>{esc(message.text)}</b>\nПотеряно: <b>{b['amount']}</b> коинов")
        except Exception as e:
            logger.warning("Не удалось уведомить игрока %s о матче %s: %s", b["user_id"], mid, e)
    for eb in event_bets_done:
        try:
            choice = eb["option1"] if eb["pick"] == "1" else eb["option2"]
            if eb["status"] == "refunded":
                await bot.send_message(eb["user_id"], f"↩️ Ставка на событие аннулирована и возвращена.\n\nВы сейчас состоите в одной из команд матча <b>{esc(m['team1'])} — {esc(m['team2'])}</b>.\nВозврат: <b>{eb['amount']}</b> коинов")
            elif eb["status"] == "won":
                await bot.send_message(eb["user_id"], f"✅ Ваша ставка на событие выиграла!\n\nМатч: <b>{esc(m['team1'])} — {esc(m['team2'])}</b>\nСобытие: <b>{esc(eb['title'])}</b>\nВыбор: <b>{esc(choice)}</b>\nСчёт: <b>{esc(message.text)}</b>\nВыигрыш: <b>{money(eb['payout'])}</b> коинов")
            else:
                await bot.send_message(eb["user_id"], f"❌ Ваша ставка на событие проиграла.\n\nМатч: <b>{esc(m['team1'])} — {esc(m['team2'])}</b>\nСобытие: <b>{esc(eb['title'])}</b>\nВыбор: <b>{esc(choice)}</b>\nСчёт: <b>{esc(message.text)}</b>\nПотеряно: <b>{eb['amount']}</b> коинов")
        except Exception as e:
            logger.warning("Не удалось уведомить игрока %s о событии матча %s: %s", eb["user_id"], mid, e)
    await state.clear()
    await message.answer(f"✅ Матч #{mid} завершён. Исход: <b>{result}</b>", reply_markup=admin_kb())
    await log_admin(f"✅ <b>Матч завершён</b>\n#{mid} {esc(m['team1'])} — {esc(m['team2'])}\nСчёт: <b>{esc(message.text)}</b>\nИсход: <b>{result}</b>")

async def delete_match_and_refund(target: Message | CallbackQuery, mid: int):
    with db() as con:
        begin_immediate(con); cur = con.cursor()
        cur.execute("SELECT * FROM matches WHERE id=?", (mid,)); m = cur.fetchone()
        if not m:
            con.rollback(); await safe_edit_or_answer(target, "❌ Матч не найден.", admin_kb()); return
        cur.execute("SELECT * FROM bets WHERE match_id=? AND status='pending'", (mid,)); bets = cur.fetchall()
        cur.execute("SELECT * FROM event_bets WHERE match_id=? AND status='pending'", (mid,)); event_bets = cur.fetchall()
        for b in bets:
            cur.execute("UPDATE bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
            if cur.rowcount:
                change_balance(cur, b["user_id"], int(b["amount"]), "match_deleted_refund", "bet", b["id"])
        for b in event_bets:
            cur.execute("UPDATE event_bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
            if cur.rowcount:
                change_balance(cur, b["user_id"], int(b["amount"]), "match_deleted_refund", "event_bet", b["id"])
        cur.execute("UPDATE match_events SET status='deleted' WHERE match_id=? AND status='active'", (mid,))
        cur.execute("UPDATE matches SET status='deleted' WHERE id=?", (mid,)); con.commit()
    await settle_miniapp_slips_for_match(mid, cancelled=True)
    for b in list(bets) + list(event_bets):
        try:
            await bot.send_message(b["user_id"], f"↩️ Матч <b>{esc(m['team1'])} — {esc(m['team2'])}</b> был удалён администратором.\nВам возвращено: <b>{b['amount']}</b> коинов.")
        except Exception as e:
            logger.warning("Не удалось уведомить о возврате матча %s пользователя %s: %s", mid, b["user_id"], e)
    total = len(bets) + len(event_bets)
    await safe_edit_or_answer(target, f"🗑 Матч #{mid} удалён. Возвращено ставок: <b>{total}</b>.", admin_kb())
    await log_admin(f"🗑 <b>Матч удалён</b>\n#{mid} {esc(m['team1'])} — {esc(m['team2'])}\nВозвращено ставок: <b>{total}</b>")

@router.message(TournamentCreate.teams)
async def st_tour_teams(message: Message, state: FSMContext):
    if not is_admin_user(message.from_user.id): return
    teams, bad, seen = [], [], set()
    for line in message.text.splitlines():
        if not line.strip(): continue
        if "|" not in line:
            bad.append(line); continue
        name, odds_s = [x.strip() for x in line.split("|", 1)]
        name = canonical_registry_club(name)
        odds = parse_float(odds_s)
        key = normalize_club_key(name)
        if not name or not odds or odds <= 0 or key in seen:
            bad.append(line); continue
        seen.add(key); teams.append((name, odds))
    if bad or not teams:
        await message.answer("❌ Ошибка формата или повтор команды. Используйте:\n<code>France | 3.5\nBrazil | 4.0</code>"); return
    data = await state.get_data()
    if int(data.get("max_picks", 1)) > len(teams):
        await message.answer(f"❌ Лимит выбора не может быть больше количества команд ({len(teams)})."); return
    await state.update_data(teams=teams)
    preview = "\n".join([f"• <b>{esc(n)}</b> — {o}" for n, o in teams])
    await state.set_state(TournamentCreate.confirm)
    await message.answer(
        "<b>📋 Предпросмотр режима победителя</b>\n\n"
        f"🏆 Лига: <b>{esc(data['league_name'])}</b>\n"
        f"🔗 Ссылка: {esc(data['league_url'])}\n"
        f"📌 Лимит: <b>{data['max_picks']}</b>\n\n{preview}",
        reply_markup=confirm_cancel_kb("tour:create_ok", "tour:create_cancel")
    )

async def settle_tournament(target: Message | CallbackQuery, tid: int, team_id: int):
    with db() as con:
        begin_immediate(con); cur = con.cursor()
        cur.execute("SELECT * FROM tournaments WHERE id=? AND status IN ('active','closed')", (tid,)); t = cur.fetchone()
        cur.execute("SELECT * FROM tournament_teams WHERE id=? AND tournament_id=? AND active=1", (team_id, tid)); winner = cur.fetchone()
        if not t or not winner:
            con.rollback(); await safe_edit_or_answer(target, "❌ Турнир или команда не найдены.", admin_kb()); return
        sync_ok, sync_detail, _ = touchbet_registry_health()
        if STRICT_CLUB_CHECK and not sync_ok:
            con.rollback(); await safe_edit_or_answer(target, f"⚠️ Турнир пока не рассчитан: нет свежей синхронизации с Transfer Markt.\n\n{esc(sync_detail)}", admin_kb()); return
        cur.execute("UPDATE tournaments SET status='finished', winner_team_id=? WHERE id=? AND status IN ('active','closed')", (team_id, tid))
        if cur.rowcount != 1:
            con.rollback(); await safe_edit_or_answer(target, "❌ Турнир уже рассчитан.", admin_kb()); return
        cur.execute("SELECT * FROM winner_bets WHERE tournament_id=? AND status='pending'", (tid,)); bets = cur.fetchall()
        tm_refunded_winner_ids = set()
        for b in bets:
            allowed_now, _ = check_player_club_for_teams(b["user_id"], b["team_name"])
            if not allowed_now:
                change_balance(cur, b["user_id"], int(b["amount"]), "tm_own_team_refund", "winner_bet", b["id"])
                cur.execute("UPDATE winner_bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
                tm_refunded_winner_ids.add(int(b["id"]))
                continue
            if b["team_id"] == team_id:
                payout = calc_payout(b["amount"], b["odds"]); profit = payout - int(b["amount"])
                change_balance(cur, b["user_id"], payout, "tournament_bet_win", "winner_bet", b["id"], earned_delta=profit)
                cur.execute("UPDATE winner_bets SET status='won', payout=? WHERE id=? AND status='pending'", (payout, b["id"]))
            else:
                cur.execute("UPDATE winner_bets SET status='lost', payout=0 WHERE id=? AND status='pending'", (b["id"],))
        con.commit()
    for b in bets:
        try:
            if int(b["id"]) in tm_refunded_winner_ids:
                await bot.send_message(b["user_id"], f"↩️ Ставка на команду <b>{esc(b['team_name'])}</b> аннулирована и возвращена, потому что по данным TM это ваша текущая команда.\nВозврат: <b>{b['amount']}</b> коинов")
            elif b["team_id"] == team_id:
                payout = calc_payout(b["amount"], b["odds"])
                await bot.send_message(b["user_id"], f"✅ Ваша ставка на победителя выиграла!\n\nТурнир: <b>{esc(t['league_name'])}</b>\nПобедитель: <b>{esc(winner['name'])}</b>\nВыигрыш: <b>{payout}</b> коинов")
            else:
                await bot.send_message(b["user_id"], f"❌ Ваша ставка на победителя проиграла.\n\nТурнир: <b>{esc(t['league_name'])}</b>\nПобедитель: <b>{esc(winner['name'])}</b>\nПотеряно: <b>{b['amount']}</b> коинов")
        except Exception as e:
            logger.warning("Не удалось уведомить о турнире %s пользователя %s: %s", tid, b["user_id"], e)
    await safe_edit_or_answer(target, f"🥇 Победитель турнира #{tid}: <b>{esc(winner['name'])}</b>", admin_kb())
    await log_admin(f"🥇 <b>Победитель турнира назначен</b>\nТурнир: <b>{esc(t['league_name'])}</b>\nПобедитель: <b>{esc(winner['name'])}</b>")

def parse_ban_time(s: str) -> Optional[str]:
    s = s.lower()
    if s == "perm":
        return "perm"
    if len(s) < 2:
        return None
    num, unit = s[:-1], s[-1]
    if not num.isdigit():
        return None
    n = int(num)
    if unit == "h":
        return (local_now() + timedelta(hours=n)).strftime("%Y-%m-%d %H:%M:%S")
    if unit == "d":
        return (local_now() + timedelta(days=n)).strftime("%Y-%m-%d %H:%M:%S")
    return None

def _tm_fixture_datetime(value: Any) -> Optional[str]:
    raw = str(value or "").strip()
    if not raw: return None
    try:
        dt = datetime.fromisoformat(raw.replace("Z", "+00:00"))
        if dt.tzinfo is None: dt = dt.replace(tzinfo=ZoneInfo("Europe/Moscow"))
        dt = dt.astimezone(LOCAL_TZ).replace(tzinfo=None)
        return dt.strftime("%Y-%m-%d %H:%M:%S")
    except Exception: return None

def _tm_event_draft(match_row: Any, generation: int = 0) -> list[dict[str, Any]]:
    fid = str(match_row["tm_fixture_id"] if isinstance(match_row, sqlite3.Row) else match_row.get("tm_fixture_id") or match_row.get("id") or "")
    team1 = str(match_row["team1"] if isinstance(match_row, sqlite3.Row) else match_row.get("team1") or "Команда 1")
    team2 = str(match_row["team2"] if isinstance(match_row, sqlite3.Row) else match_row.get("team2") or "Команда 2")
    seed = int(hashlib.sha256(f"{fid}:{generation}".encode()).hexdigest()[:12], 16)
    totals = [6.5, 7.5, 8.5, 9.5, 10.5, 11.5]; total = totals[seed % len(totals)]
    it_base = max(2.5, round((total / 2.0) * 2) / 2); shifts = [-0.5, 0.0, 0.5]
    it1 = max(1.5, it_base + shifts[(seed // 7) % 3]); it2 = max(1.5, it_base + shifts[(seed // 13) % 3])
    odds_pool = [1.75, 1.80, 1.85, 1.90, 1.95, 2.00]
    def pair(n): return odds_pool[(seed // (n + 3)) % len(odds_pool)], odds_pool[(seed // (n + 9) + 2) % len(odds_pool)]
    try:
        mo1 = float(match_row["odds1"] if isinstance(match_row, sqlite3.Row) else match_row.get("odds1") or 0)
        mo2 = float(match_row["odds2"] if isinstance(match_row, sqlite3.Row) else match_row.get("odds2") or 0)
    except Exception:
        mo1 = mo2 = 0.0
    if mo1 > 0 and mo2 > 0 and abs(mo1 - mo2) >= 0.08:
        handicap_kind = "handicap1" if mo1 < mo2 else "handicap2"
        handicap_line = [-2.5, -1.5][(seed // 19) % 2]
    else:
        handicap_kind = "handicap1" if (seed // 17) % 2 == 0 else "handicap2"
        handicap_line = [-1.5, 1.5][(seed // 19) % 2]
    specs = [("total", total), ("it1", it1), ("it2", it2), (handicap_kind, handicap_line)]
    out=[]
    for i,(kind,line) in enumerate(specs):
        o1,o2=pair(i); payload=create_event_payload(kind,float(line),o1,o2,team1,team2)
        out.append({"kind":kind,"line":float(line),"odds1":o1,"odds2":o2,**payload})
    return out


def _tm_team_rating(name: str) -> tuple[float, Optional[int], str]:
    """Read a stable club strength exported by TM. Unknown teams stay neutral."""
    roster, _ = _load_external_roster()
    canonical = canonical_registry_club(name)
    ratings = (roster or {}).get("team_ratings") or {}
    item = ratings.get(canonical) if isinstance(ratings, dict) else None
    if isinstance(item, dict):
        try:
            rating = float(item.get("rating") or 0)
        except Exception:
            rating = 0.0
        try:
            rank = int(item.get("cis_rank")) if item.get("cis_rank") is not None else None
        except Exception:
            rank = None
        if rating > 0:
            return rating, rank, canonical
    # If one/both sides are not in TOP CIS, do not invent a large disadvantage.
    return 1400.0, None, canonical


def _tm_auto_match_odds(team1: str, team2: str, fixture_id: str = "", generation: int = 0) -> tuple[float, float, float, str]:
    """Generate transparent 1/X/2 odds from TM ranking with a small bookmaker margin."""
    r1, rank1, c1 = _tm_team_rating(team1)
    r2, rank2, c2 = _tm_team_rating(team2)
    seed = int(hashlib.sha256(f"odds:{fixture_id}:{generation}".encode()).hexdigest()[:10], 16)
    # Tiny deterministic variation is only for regenerate; ranking remains dominant.
    jitter = ((seed % 21) - 10) * 1.5 if generation else 0.0
    delta = (r1 - r2) + jitter
    # Draws are more likely for evenly matched teams.
    draw_p = 0.18 + 0.08 * math.exp(-abs(delta) / 320.0)
    draw_p = min(0.27, max(0.15, draw_p))
    decisive = 1.0 - draw_p
    p1_share = 1.0 / (1.0 + math.exp(-delta / 190.0))
    p1 = decisive * p1_share
    p2 = decisive - p1
    # 7% overround, then sensible caps for a small virtual market.
    margin = 1.07
    def odd(p: float) -> float:
        return round(min(8.50, max(1.12, 1.0 / max(0.02, p * margin))), 2)
    o1, ox, o2 = odd(p1), odd(draw_p), odd(p2)
    if rank1 and rank2:
        why = f"TM ТОП СНГ: #{rank1} {c1} vs #{rank2} {c2}"
    elif rank1:
        why = f"TM ТОП СНГ: #{rank1} {c1}; соперник без позиции"
    elif rank2:
        why = f"TM ТОП СНГ: #{rank2} {c2}; соперник без позиции"
    else:
        why = "нейтральная модель: обе команды без позиции в ТОП СНГ"
    return o1, ox, o2, why


def _tm_apply_auto_odds(cur: sqlite3.Cursor, row: sqlite3.Row, generation: int, source: str = "auto") -> tuple[float, float, float, str]:
    o1, ox, o2, why = _tm_auto_match_odds(row["team1"], row["team2"], str(row["tm_fixture_id"] or row["id"]), generation)
    cur.execute(
        "UPDATE matches SET odds1=?, oddsx=?, odds2=?, tm_odds_generation=?, tm_odds_source=? WHERE id=?",
        (o1, ox, o2, int(generation), source, row["id"]),
    )
    return o1, ox, o2, why

async def _tm_upgrade_pending_auto_odds() -> int:
    """One-time safe migration: only unconfirmed V4.1 rows get auto odds."""
    changed = []
    with db() as con:
        begin_immediate(con); cur = con.cursor()
        cur.execute("SELECT * FROM matches WHERE status='tm_pending' AND COALESCE(tm_odds_source,'')='' ORDER BY id")
        rows = cur.fetchall()
        for row in rows:
            _tm_apply_auto_odds(cur, row, int(row["tm_odds_generation"] or 0), "auto")
            cur.execute("SELECT * FROM matches WHERE id=?", (row["id"],)); fresh = cur.fetchone()
            draft = _tm_event_draft(fresh, int(fresh["tm_generation"] or 0))
            cur.execute("UPDATE matches SET tm_event_draft_json=? WHERE id=?", (json.dumps(draft, ensure_ascii=False), row["id"]))
            changed.append((int(row["id"]), int(row["tm_sync_log_msg_id"] or 0)))
        con.commit()
    for mid, log_id in changed:
        with db() as con:
            cur=con.cursor(); cur.execute("SELECT * FROM matches WHERE id=?", (mid,)); row=cur.fetchone()
        if not row:
            continue
        if log_id:
            try:
                await bot.edit_message_text(_tm_sync_preview(row), chat_id=LOG_GROUP_ID, message_id=log_id, reply_markup=_tm_sync_log_kb(mid))
                continue
            except Exception:
                logger.warning("Could not edit old TM preview %s; sending a fresh preview", mid)
        await _tm_send_sync_preview(mid)
    return len(changed)

async def _tm_publish_match(mid:int,actor_id:int):
    with db() as con:
        begin_immediate(con); cur=con.cursor(); cur.execute("SELECT * FROM matches WHERE id=?",(mid,)); m=cur.fetchone()
        if not m or m["status"]!="tm_pending": con.rollback(); return False,"Матч уже обработан."
        scheduled = parse_db_datetime(m["match_time"] if "match_time" in m.keys() else None)
        if scheduled and local_now() >= scheduled:
            con.rollback(); return False,"Назначенное время матча уже наступило — публиковать ставки нельзя."
        events=_tm_draft_from_row(m); cur.execute("UPDATE matches SET status='active',tm_sync_state='published',published_at=? WHERE id=? AND status='tm_pending'",(now_str(),mid))
        if cur.rowcount!=1: con.rollback(); return False,"Матч уже обработан."
        cur.execute("SELECT COUNT(*) FROM match_events WHERE match_id=?",(mid,)); already=cur.fetchone()[0]
        if not already:
            for ev in events: cur.execute("INSERT INTO match_events (match_id,kind,title,option1,option2,odds1,odds2,line,team_side,status,created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'active', ?)",(mid,ev.get("kind"),ev.get("title"),ev.get("option1"),ev.get("option2"),float(ev.get("odds1")),float(ev.get("odds2")),ev.get("line"),ev.get("team_side"),now_str()))

    _tm_write_bridge(); return True,"Матч опубликован игрокам."

def _tm_parse_score(value: Any) -> Optional[tuple[int, int]]:
    m = re.fullmatch(r"\s*(\d+)\s*[:\-]\s*(\d+)\s*", str(value or ""))
    if not m:
        return None
    return int(m.group(1)), int(m.group(2))


async def _tm_cancel_from_registry(mid: int, fixture: dict[str, Any]) -> None:
    """Cancel a synchronized TouchBet match and atomically refund open bets."""
    with db() as con:
        begin_immediate(con)
        cur = con.cursor()
        cur.execute("SELECT * FROM matches WHERE id=?", (mid,))
        m = cur.fetchone()
        if not m or m["status"] in {"deleted", "finished"}:
            con.rollback()
            return
        cur.execute("SELECT * FROM bets WHERE match_id=? AND status='pending'", (mid,))
        regular = cur.fetchall()
        cur.execute("SELECT * FROM event_bets WHERE match_id=? AND status='pending'", (mid,))
        event_rows = cur.fetchall()
        for b in regular:
            cur.execute("UPDATE bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
            if cur.rowcount:
                change_balance(cur, b["user_id"], int(b["amount"]), "tm_fixture_cancel_refund", "bet", b["id"])
        for b in event_rows:
            cur.execute("UPDATE event_bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
            if cur.rowcount:
                change_balance(cur, b["user_id"], int(b["amount"]), "tm_fixture_cancel_refund", "event_bet", b["id"])
        cur.execute("UPDATE match_events SET status='deleted' WHERE match_id=? AND status='active'", (mid,))
        cur.execute("UPDATE matches SET status='deleted', tm_sync_state='cancelled_by_tm' WHERE id=?", (mid,))
        cur.execute("UPDATE tm_notify_queue SET sent=1, sent_at=? WHERE match_id=? AND sent=0", (now_str(), mid))
        con.commit()
    await settle_miniapp_slips_for_match(mid, cancelled=True)
    affected = sorted({int(b["user_id"]) for b in list(regular) + list(event_rows)})
    for uid in affected:
        try:
            await bot.send_message(uid, f"↩️ <b>Матч отменён в Transfer Markt</b>\n\n{esc(m['team1'])} — {esc(m['team2'])}\nВсе незавершённые ставки на этот матч возвращены.")
        except Exception:
            pass
    await log_admin(f"↩️ <b>TM sync: матч отменён</b>\n#{mid} {esc(m['team1'])} — {esc(m['team2'])}\nВозвращено ставок: <b>{len(regular)+len(event_rows)}</b>")


async def _tm_finish_from_registry(mid: int, fixture: dict[str, Any]) -> None:
    """Use the official TM result to settle a synchronized TouchBet match once."""
    parsed = _tm_parse_score(fixture.get("score"))
    if not parsed:
        return
    g1, g2 = parsed
    result = "1" if g1 > g2 else "2" if g2 > g1 else "X"
    with db() as con:
        begin_immediate(con)
        cur = con.cursor()
        cur.execute("SELECT * FROM matches WHERE id=?", (mid,))
        m = cur.fetchone()
        if not m or m["status"] in {"finished", "deleted"}:
            con.rollback()
            return
        cur.execute("SELECT * FROM bets WHERE match_id=? AND status='pending'", (mid,))
        regular = cur.fetchall()
        if m["format"] == "playoff" and result == "X":
            con.rollback()
            return
        cur.execute("UPDATE matches SET status='finished', result=?, score=?, tm_sync_state='finished_by_tm' WHERE id=? AND status IN ('active','tm_pending')", (result, f"{g1}:{g2}", mid))
        if cur.rowcount != 1:
            con.rollback()
            return
        tm_refunded_regular_ids = set()
        for b in regular:
            allowed_now, _ = check_player_club_for_teams(b["user_id"], m["team1"], m["team2"])
            if not allowed_now:
                change_balance(cur, b["user_id"], int(b["amount"]), "tm_own_team_refund", "bet", b["id"])
                cur.execute("UPDATE bets SET status='refunded', payout=? WHERE id=? AND status='pending'", (b["amount"], b["id"]))
                tm_refunded_regular_ids.add(int(b["id"]))
                continue
            if b["pick"] == result:
                payout = calc_payout(b["amount"], b["odds"])
                profit = payout - int(b["amount"])
                change_balance(cur, b["user_id"], payout, "tm_synced_bet_win", "bet", b["id"], earned_delta=profit)
                cur.execute("UPDATE bets SET status='won', payout=? WHERE id=? AND status='pending'", (payout, b["id"]))
            else:
                cur.execute("UPDATE bets SET status='lost', payout=0 WHERE id=? AND status='pending'", (b["id"],))
        cur.execute("SELECT id FROM event_bets WHERE match_id=? AND status='pending'", (mid,))
        pending_event_ids = [int(r[0]) for r in cur.fetchall()]
        await settle_events_for_match(cur, mid, g1, g2)
        cur.execute("UPDATE tm_notify_queue SET sent=1, sent_at=? WHERE match_id=? AND sent=0", (now_str(), mid))
        con.commit()
    await settle_miniapp_slips_for_match(mid, g1, g2)
    users = sorted({int(b["user_id"]) for b in regular})
    if pending_event_ids:
        with db() as con:
            cur = con.cursor()
            ph = ",".join("?" for _ in pending_event_ids)
            cur.execute(f"SELECT DISTINCT user_id FROM event_bets WHERE id IN ({ph})", pending_event_ids)
            users = sorted(set(users) | {int(r[0]) for r in cur.fetchall()})
    for uid in users:
        try:
            await bot.send_message(uid, f"✅ <b>Матч завершён в Transfer Markt</b>\n\n{esc(m['team1'])} <b>{g1}:{g2}</b> {esc(m['team2'])}\nTouchBet автоматически рассчитал ваши ставки.")
        except Exception:
            pass
    await log_admin(f"✅ <b>TM sync: матч рассчитан</b>\n#{mid} {esc(m['team1'])} <b>{g1}:{g2}</b> {esc(m['team2'])}\nОбычных ставок: <b>{len(regular)}</b> • event bets: <b>{len(pending_event_ids)}</b>")

def _mini_validate_init_data(raw: str, max_age: int):
    try:
        pairs = dict(parse_qsl(str(raw or ""), keep_blank_values=True))
        received = str(pairs.pop("hash", "") or "")
        if not received:
            return None, "missing_hash"
        # Telegram Mini Apps: secret = HMAC_SHA256(key='WebAppData', msg=bot_token)
        secret = hmac.new(b"WebAppData", BOT_TOKEN.encode("utf-8"), hashlib.sha256).digest()
        check = "\n".join(f"{k}={pairs[k]}" for k in sorted(pairs))
        calculated = hmac.new(secret, check.encode("utf-8"), hashlib.sha256).hexdigest()
        if not hmac.compare_digest(calculated, received):
            return None, "bad_hash"
        auth_date = int(pairs.get("auth_date") or 0)
        if auth_date <= 0 or abs(int(time.time()) - auth_date) > int(max_age):
            return None, "expired_init_data"
        user = json.loads(pairs.get("user") or "{}")
        if not isinstance(user, dict) or not user.get("id"):
            return None, "missing_user"
        return user, None
    except Exception:
        return None, "invalid_init_data"

def _mini_decimal_odds(values) -> float:
    value = Decimal("1")
    for x in values:
        value *= Decimal(str(x))
    return float(value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))


def _mini_price_probability(prob: float, margin: float = 1.06, low: float = 1.05, high: float = 8.5) -> float:
    prob = max(0.02, min(0.96, float(prob)))
    return round(max(low, min(high, 1.0 / (prob * margin))), 2)


def _mini_base_probs(m: sqlite3.Row) -> tuple[float,float,float]:
    vals = []
    for key in ("odds1","oddsx","odds2"):
        try:
            odd = float(m[key] or 0)
        except Exception:
            odd = 0.0
        vals.append(1.0 / odd if odd > 1.0 else 0.0)
    if vals[1] <= 0:
        vals[1] = 0.0
    s = sum(vals)
    if s <= 0:
        return 0.40, 0.20, 0.40
    return vals[0]/s, vals[1]/s, vals[2]/s


def _mini_selection(match_id: int, selection_id: str, market_key: str, market_title: str, pick: str, label: str, odds: float,
                    event_kind: str | None = None, line: float | None = None, team_side: str | None = None):
    return {
        "match_id": int(match_id), "selection_id": str(selection_id), "market_key": market_key,
        "market_title": market_title, "pick": pick, "label": label, "odds": float(odds),
        "event_kind": event_kind, "line": line, "team_side": team_side,
    }


def _mini_market_catalog(m: sqlite3.Row) -> list[dict[str,Any]]:
    mid = int(m["id"])
    groups = []
    p1, px, p2 = _mini_base_probs(m)
    one = [
        _mini_selection(mid, f"m{mid}:1", "1x2", "Исход матча", "1", f"П1 · {m['team1']}", float(m["odds1"])),
        _mini_selection(mid, f"m{mid}:2", "1x2", "Исход матча", "2", f"П2 · {m['team2']}", float(m["odds2"])),
    ]
    if m["format"] == "league" and m["oddsx"] is not None:
        one.insert(1, _mini_selection(mid, f"m{mid}:X", "1x2", "Исход матча", "X", "Ничья", float(m["oddsx"])))
    groups.append({"key":"1x2","title":"Исход","options":one})

    if m["format"] == "league":
        dc = [
            _mini_selection(mid, f"m{mid}:dc:1X", "double_chance", "Двойной шанс", "1X", "1X", _mini_price_probability(p1+px,1.045,1.05,4.0)),
            _mini_selection(mid, f"m{mid}:dc:12", "double_chance", "Двойной шанс", "12", "12", _mini_price_probability(p1+p2,1.045,1.05,4.0)),
            _mini_selection(mid, f"m{mid}:dc:X2", "double_chance", "Двойной шанс", "X2", "X2", _mini_price_probability(px+p2,1.045,1.05,4.0)),
        ]
        groups.append({"key":"double_chance","title":"Двойной шанс","options":dc})

    with db() as con:
        cur = con.cursor(); cur.execute("SELECT * FROM match_events WHERE match_id=? AND status='active' ORDER BY id", (mid,)); events = cur.fetchall()
    total_line = None
    for ev in events:
        if ev["kind"] == "total":
            total_line = float(ev["line"] or 0); break
    # High-scoring football simulations should not inherit real-world 2.5 assumptions.
    if total_line is None:
        total_line = 5.5
    yes_p = max(0.48, min(0.86, 0.56 + max(0.0, total_line-2.5)*0.035))
    btts = [
        _mini_selection(mid, f"m{mid}:btts:Y", "btts", "Обе забьют", "Y", "Да", _mini_price_probability(yes_p,1.06,1.08,4.5)),
        _mini_selection(mid, f"m{mid}:btts:N", "btts", "Обе забьют", "N", "Нет", _mini_price_probability(1.0-yes_p,1.06,1.08,5.5)),
    ]
    groups.append({"key":"btts","title":"Обе забьют","options":btts})

    for ev in events:
        kind = str(ev["kind"] or "")
        if kind not in {"total","it1","it2"}:
            continue
        opts = [
            _mini_selection(mid, f"e{ev['id']}:1", f"event:{ev['id']}", str(ev["title"]), "1", str(ev["option1"]), float(ev["odds1"]), kind, float(ev["line"] or 0), ev["team_side"]),
            _mini_selection(mid, f"e{ev['id']}:2", f"event:{ev['id']}", str(ev["title"]), "2", str(ev["option2"]), float(ev["odds2"]), kind, float(ev["line"] or 0), ev["team_side"]),
        ]
        groups.append({"key":f"event:{ev['id']}","title":str(ev["title"]),"options":opts})
    return groups


def _mini_find_selection(m: sqlite3.Row, selection_id: str):
    for group in _mini_market_catalog(m):
        for option in group["options"]:
            if option["selection_id"] == str(selection_id):
                return option
    return None


def _mini_eval_leg(leg: sqlite3.Row, g1: int, g2: int) -> bool:
    key = str(leg["market_key"] or "")
    pick = str(leg["pick"] or "")
    if key == "1x2":
        result = "1" if g1 > g2 else "2" if g2 > g1 else "X"
        return pick == result
    if key == "double_chance":
        result = "1" if g1 > g2 else "2" if g2 > g1 else "X"
        return result in set(pick)
    if key == "btts":
        yes = g1 > 0 and g2 > 0
        return (pick == "Y" and yes) or (pick == "N" and not yes)
    kind = str(leg["event_kind"] or "")
    line = float(leg["line"] or 0)
    if kind == "total":
        result = "1" if (g1+g2) > line else "2"
    elif kind == "it1":
        result = "1" if g1 > line else "2"
    elif kind == "it2":
        result = "1" if g2 > line else "2"
    else:
        return False
    return pick == result


def _mini_finalize_slip(cur: sqlite3.Cursor, slip_id: int):
    cur.execute("SELECT * FROM bet_slips WHERE id=?", (int(slip_id),)); slip = cur.fetchone()
    if not slip or slip["status"] != "pending":
        return None
    cur.execute("SELECT * FROM bet_legs WHERE slip_id=? ORDER BY id", (int(slip_id),)); legs = cur.fetchall()
    statuses = [str(x["status"]) for x in legs]
    if any(x == "lost" for x in statuses):
        cur.execute("UPDATE bet_slips SET status='lost',payout=0,settled_at=? WHERE id=? AND status='pending'", (now_str(),slip_id))
        return {"user_id":slip["user_id"],"slip_id":slip_id,"status":"lost","payout":0}
    if any(x == "pending" for x in statuses):
        return None
    won = [x for x in legs if x["status"] == "won"]
    if not won:
        payout = int(slip["stake"]); status = "refunded"; earned = 0
    else:
        odds = _mini_decimal_odds([float(x["odds"]) for x in won])
        payout = calc_payout(int(slip["stake"]), odds)
        status = "won"; earned = payout - int(slip["stake"])
    cur.execute("UPDATE bet_slips SET status=?,payout=?,settled_at=? WHERE id=? AND status='pending'", (status,payout,now_str(),slip_id))
    if cur.rowcount == 1:
        change_balance(cur, int(slip["user_id"]), int(payout), "miniapp_slip_"+status, "bet_slip", slip_id, earned_delta=earned)
        return {"user_id":slip["user_id"],"slip_id":slip_id,"status":status,"payout":payout}
    return None


async def settle_miniapp_slips_for_match(match_id: int, g1: int | None = None, g2: int | None = None, cancelled: bool = False):
    notices = []
    with db() as con:
        begin_immediate(con); cur = con.cursor()
        cur.execute("SELECT DISTINCT slip_id FROM bet_legs WHERE match_id=? AND status='pending'", (int(match_id),)); slip_ids = [int(r[0]) for r in cur.fetchall()]
        if not slip_ids:
            con.rollback(); return 0
        cur.execute("SELECT * FROM bet_legs WHERE match_id=? AND status='pending'", (int(match_id),)); legs = cur.fetchall()
        cur.execute("SELECT team1,team2 FROM matches WHERE id=?", (int(match_id),)); match_row = cur.fetchone()
        for leg in legs:
            if cancelled:
                cur.execute("UPDATE bet_legs SET status='refunded',result='cancelled' WHERE id=? AND status='pending'", (leg["id"],))
            elif match_row and not check_player_club_for_teams(int(leg["user_id"]), match_row["team1"], match_row["team2"])[0]:
                # If the player joined/owns one of the teams after placing the bet,
                # this leg becomes void. In an accumulator, void legs are odds 1.00.
                cur.execute("UPDATE bet_legs SET status='refunded',result='tm_own_team' WHERE id=? AND status='pending'", (leg["id"],))
            else:
                won = _mini_eval_leg(leg, int(g1 or 0), int(g2 or 0))
                cur.execute("UPDATE bet_legs SET status=?,result=? WHERE id=? AND status='pending'", ("won" if won else "lost", f"{int(g1 or 0)}:{int(g2 or 0)}", leg["id"]))
        for sid in slip_ids:
            result = _mini_finalize_slip(cur, sid)
            if result: notices.append(result)
        con.commit()
    for n in notices:
        try:
            if n["status"] == "won":
                await bot.send_message(int(n["user_id"]), f"✅ <b>Ставка TouchBet выиграла</b>\n\nКупон #{n['slip_id']}\n💰 Выплата: <b>{n['payout']}</b> коинов")
            elif n["status"] == "refunded":
                await bot.send_message(int(n["user_id"]), f"↩️ <b>Купон TouchBet возвращён</b>\n\nКупон #{n['slip_id']}\n💰 Возврат: <b>{n['payout']}</b> коинов")
            else:
                await bot.send_message(int(n["user_id"]), f"❌ <b>Купон TouchBet проиграл</b>\n\nКупон #{n['slip_id']}")
        except Exception:
            pass
    return len(slip_ids)


def _mini_place_slip(uid: int, stake: int, selection_ids: list[str]):
    stake = int(stake)
    if stake < MIN_BET:
        return False, f"Минимальная ставка — {MIN_BET} коинов.", None
    ids = [str(x or "").strip() for x in selection_ids if str(x or "").strip()]
    if not ids:
        return False, "Купон пуст.", None
    if len(ids) > MINIAPP_MAX_EXPRESS_LEGS:
        return False, f"Максимум {MINIAPP_MAX_EXPRESS_LEGS} исходов в экспрессе.", None
    if len(set(ids)) != len(ids):
        return False, "В купоне есть повторяющиеся исходы.", None
    selected = []
    with db() as con:
        cur = con.cursor()
        for sid in ids:
            m = re.match(r"^[me](\d+)", sid)
            if not m:
                return False, "Некорректный исход в купоне.", None
            if sid.startswith("m"):
                mid = int(m.group(1))
            else:
                eid = int(m.group(1)); cur.execute("SELECT match_id FROM match_events WHERE id=?", (eid,)); rr=cur.fetchone()
                if not rr: return False, "Событие больше недоступно.", None
                mid = int(rr[0])
            cur.execute("SELECT * FROM matches WHERE id=? AND status='active'", (mid,)); match = cur.fetchone()
            if not match or match_is_closed_for_bets(match):
                return False, "Один из матчей уже закрыт для ставок.", None
            allowed, reason = check_player_club_for_teams(uid, match["team1"], match["team2"])
            if not allowed:
                return False, re.sub(r"<[^>]+>", "", reason), None
            opt = _mini_find_selection(match, sid)
            if not opt:
                return False, "Один из коэффициентов изменился или рынок закрыт. Обновите купон.", None
            opt["team1"] = match["team1"]; opt["team2"] = match["team2"]
            selected.append(opt)
    match_ids = [x["match_id"] for x in selected]
    if len(selected) > 1 and len(set(match_ids)) != len(match_ids):
        return False, "В экспрессе можно выбрать только один исход из каждого матча.", None
    total_odds = _mini_decimal_odds([x["odds"] for x in selected])
    potential = calc_payout(stake, total_odds)
    with db() as con:
        begin_immediate(con); cur = con.cursor()
        cur.execute("SELECT balance FROM users WHERE id=?", (int(uid),)); row=cur.fetchone()
        if not row or int(row[0]) < stake:
            con.rollback(); return False, "Недостаточно коинов.", None
        kind = "single" if len(selected) == 1 else "express"
        cur.execute("INSERT INTO bet_slips (user_id,kind,stake,total_odds,potential_payout,status,payout,source,created_at) VALUES (?,?,?,?,?,'pending',0,'miniapp',?)",
                    (int(uid),kind,stake,total_odds,potential,now_str()))
        slip_id = int(cur.lastrowid)
        for x in selected:
            cur.execute("INSERT INTO bet_legs (slip_id,user_id,match_id,selection_id,market_key,market_title,pick,label,odds,event_kind,line,team_side,status,created_at) VALUES (?,?,?,?,?,?,?,?,?,?,?,?, 'pending', ?)",
                        (slip_id,int(uid),x["match_id"],x["selection_id"],x["market_key"],x["market_title"],x["pick"],x["label"],x["odds"],x.get("event_kind"),x.get("line"),x.get("team_side"),now_str()))
        ok,_ = change_balance(cur,int(uid),-stake,"miniapp_slip_stake","bet_slip",slip_id)
        if not ok:
            con.rollback(); return False, "Недостаточно коинов.", None
        con.commit()
    return True, "Ставка принята.", {"id":slip_id,"kind":kind,"stake":stake,"total_odds":total_odds,"potential_payout":potential,"legs":selected}

def _tb_fetch_tm_snapshot_http():
    if not TM_API_URL or not TOUCHBET_BRIDGE_SECRET:
        return False, "TM_API_URL/TOUCHBET_BRIDGE_SECRET not configured"
    try:
        req=Request(TM_API_URL+"/api/touchbet/snapshot", headers={"Authorization":"Bearer "+TOUCHBET_BRIDGE_SECRET,"Accept":"application/json"}, method="GET")
        with urlopen(req,timeout=6) as resp:
            raw=resp.read(4_000_000)
        doc=json.loads(raw.decode("utf-8")); data=doc.get("data") if isinstance(doc,dict) else None
        if not isinstance(data,dict) or not isinstance(data.get("players"),dict):
            return False,"TM snapshot schema mismatch"
        tmp=TOUCHBET_ROSTER_FILE+".tmp"
        os.makedirs(os.path.dirname(TOUCHBET_ROSTER_FILE) or ".",exist_ok=True)
        with open(tmp,"w",encoding="utf-8") as f: json.dump(data,f,ensure_ascii=False,separators=(",",":"))
        os.replace(tmp,TOUCHBET_ROSTER_FILE)
        _ROSTER_CACHE.update({"mtime":None,"data":None,"error":None,"path":None})
        _TM_HTTP_LAST.update({"ok":True,"at":now_str(),"error":None})
        return True,None
    except Exception as exc:
        _TM_HTTP_LAST.update({"ok":False,"at":now_str(),"error":f"{type(exc).__name__}: {exc}"})
        return False,str(exc)


async def tm_http_snapshot_loop():
    while True:
        try:
            ok,err=await asyncio.to_thread(_tb_fetch_tm_snapshot_http)
            if not ok and err: logger.warning("TM HTTP snapshot: %s",err)
            if ok:
                try: await sync_tm_fixtures_once()
                except Exception: logger.exception("TM fixture sync after HTTP snapshot")
        except asyncio.CancelledError: raise
        except Exception: logger.exception("TM HTTP snapshot loop")
        try:
            if _TOUCHBET_TM_WAKE is not None:
                await asyncio.wait_for(_TOUCHBET_TM_WAKE.wait(), timeout=TM_HTTP_SYNC_INTERVAL)
                _TOUCHBET_TM_WAKE.clear()
            else:
                await asyncio.sleep(TM_HTTP_SYNC_INTERVAL)
        except asyncio.TimeoutError:
            pass

class _TouchBetMiniHandler(BaseHTTPRequestHandler):
    server_version = "TouchBetMini/0.1"
    def log_message(self, fmt, *args):
        return
    def _send(self,status,body,ctype="application/json; charset=utf-8"):
        raw=body if isinstance(body,(bytes,bytearray)) else str(body).encode("utf-8")
        self.send_response(int(status)); self.send_header("Content-Type",ctype); self.send_header("Content-Length",str(len(raw))); self.send_header("Cache-Control","no-store" if "json" in ctype else "public, max-age=60"); self.send_header("Connection","close"); self.end_headers(); self.wfile.write(raw); self.close_connection=True
    def _json(self,status,payload): self._send(status,json.dumps(payload,ensure_ascii=False,separators=(",",":")),"application/json; charset=utf-8")
    def _body(self):
        try:
            n=min(65536,max(0,int(self.headers.get("Content-Length") or 0))); return json.loads(self.rfile.read(n).decode("utf-8")) if n else {}
        except Exception: return None
    def _auth(self,write=False):
        user,err=_mini_validate_init_data(self.headers.get("X-Telegram-Init-Data", ""), MINIAPP_INITDATA_WRITE_MAX_AGE if write else MINIAPP_INITDATA_READ_MAX_AGE)
        if err: return None,err
        _mini_register_tg_user(user); banned,reason=_mini_user_banned(int(user["id"]))
        if banned: return None,"banned:"+reason
        return user,None
    def do_GET(self):
        p=urlparse(self.path); path=p.path
        if path=="/": return self._send(200,TOUCHBET_APP_HTML,"text/html; charset=utf-8")
        if path=="/app.css": return self._send(200,TOUCHBET_APP_CSS,"text/css; charset=utf-8")
        if path=="/app.js": return self._send(200,TOUCHBET_APP_JS,"application/javascript; charset=utf-8")
        if path in {"/health","/api/health"}: return self._json(200,{"ok":True,"version":TOUCHBET_MINIAPP_VERSION,"tm_http":_TM_HTTP_LAST})
        if not path.startswith("/api/"): return self._json(404,{"ok":False,"error":"not_found"})
        user,err=self._auth(False)
        if err: return self._json(401,{"ok":False,"error":err})
        uid=int(user["id"])
        if path=="/api/bootstrap":
            u=get_user(uid); matches=_mini_open_matches(uid); active=_mini_slip_rows(uid,False,100)
            return self._json(200,{"ok":True,"version":TOUCHBET_MINIAPP_VERSION,"user":{"id":uid,"name":u["first_name"],"username":u["username"],"balance":int(u["balance"]),"earned_from_bets":float(u["earned_from_bets"]),"premium":bool(u["premium"]),"active_slips":len(active),"tm_sync":bool(touchbet_registry_health()[0])},"matches":matches})
        if path=="/api/matches": return self._json(200,{"ok":True,"data":_mini_open_matches(uid)})
        if path=="/api/match":
            qs=parse_qs(p.query); raw=(qs.get("id") or [""])[0]
            if not str(raw).isdigit(): return self._json(400,{"ok":False,"error":"bad_match_id"})
            with db() as con: cur=con.cursor(); cur.execute("SELECT * FROM matches WHERE id=? AND status='active'",(int(raw),)); m=cur.fetchone()
            if not m or match_is_closed_for_bets(m): return self._json(404,{"ok":False,"error":"match_closed"})
            allowed,_=check_player_club_for_teams(uid,m["team1"],m["team2"]); summary=_mini_match_summary(m); summary["blocked_own_team"]=not allowed
            return self._json(200,{"ok":True,"data":{"match":summary,"markets":_mini_market_catalog(m)}})
        if path=="/api/bets": return self._json(200,{"ok":True,"data":_mini_slip_rows(uid,False)})
        if path=="/api/history": return self._json(200,{"ok":True,"data":_mini_slip_rows(uid,True)})
        return self._json(404,{"ok":False,"error":"api_not_found"})
    def do_POST(self):
        p=urlparse(self.path); path=p.path
        # TM webhook only wakes the secure snapshot pull; no user action is trusted from it.
        if path=="/api/tm/event":
            auth=str(self.headers.get("Authorization") or "")
            if not TOUCHBET_BRIDGE_SECRET or not hmac.compare_digest(auth,"Bearer "+TOUCHBET_BRIDGE_SECRET): return self._json(403,{"ok":False,"error":"forbidden"})
            if _TOUCHBET_ASYNC_LOOP and _TOUCHBET_TM_WAKE is not None: _TOUCHBET_ASYNC_LOOP.call_soon_threadsafe(_TOUCHBET_TM_WAKE.set)
            return self._json(202,{"ok":True})
        user,err=self._auth(True)
        if err: return self._json(401,{"ok":False,"error":err})
        body=self._body()
        if not isinstance(body,dict): return self._json(400,{"ok":False,"error":"bad_json"})
        if path=="/api/place-slip":
            try: stake=int(body.get("stake") or 0)
            except Exception: stake=0
            sels=body.get("selections") if isinstance(body.get("selections"),list) else []
            ok,msg,data=_mini_place_slip(int(user["id"]),stake,sels)
            if not ok: return self._json(409,{"ok":False,"error":msg})
            try:
                who=f"@{user.get('username')}" if user.get('username') else str(user.get('first_name') or user['id'])
                _tb_async(log_admin(f"🎫 <b>Mini App ставка</b>\nИгрок: <b>{esc(who)}</b> | <code>{user['id']}</code>\nКупон: <b>#{data['id']}</b> • {esc(data['kind'])}\nСумма: <b>{data['stake']}</b> • Кэф: <b>{data['total_odds']}</b> • Возможная выплата: <b>{data['potential_payout']}</b>"))
            except Exception: pass
            return self._json(200,{"ok":True,"message":msg,"data":data})
        return self._json(404,{"ok":False,"error":"api_not_found"})


def start_touchbet_http():
    global _TOUCHBET_HTTP_SERVER,_TOUCHBET_HTTP_THREAD
    if not TOUCHBET_MINIAPP_ENABLED or _TOUCHBET_HTTP_SERVER is not None: return
    try:
        _TOUCHBET_HTTP_SERVER=ThreadingHTTPServer((TOUCHBET_MINIAPP_HOST,TOUCHBET_MINIAPP_PORT),_TouchBetMiniHandler)
        _TOUCHBET_HTTP_SERVER.daemon_threads=True
        _TOUCHBET_HTTP_THREAD=threading.Thread(target=_TOUCHBET_HTTP_SERVER.serve_forever,name="touchbet_mini_http",daemon=True); _TOUCHBET_HTTP_THREAD.start()
        logger.info("TouchBet Mini App 0.1 HTTP ready on %s:%s",TOUCHBET_MINIAPP_HOST,TOUCHBET_MINIAPP_PORT)
    except Exception:
        logger.exception("TouchBet Mini App HTTP start failed")


def stop_touchbet_http():
    global _TOUCHBET_HTTP_SERVER,_TOUCHBET_HTTP_THREAD
    srv=_TOUCHBET_HTTP_SERVER; _TOUCHBET_HTTP_SERVER=None
    if srv:
        try: srv.shutdown(); srv.server_close()
        except Exception: pass
    _TOUCHBET_HTTP_THREAD=None

@router.message()
async def fallback(message: Message):
    if message.text and message.text.startswith("/"):
        if not is_admin_user(message.from_user.id) and message.chat.type != ChatType.PRIVATE:
            return
        if message.chat.type == ChatType.PRIVATE:
            await message.answer("Команда не найдена. Используйте /start")
        return
    if message.chat.type != ChatType.PRIVATE and not is_admin_user(message.from_user.id):
        return

async def cleanup_expired_bans_loop():
    while True:
        try:
            with db() as con:
                con.execute("UPDATE users SET banned=0, banned_until=NULL, ban_reason=NULL WHERE banned=1 AND banned_until IS NOT NULL AND banned_until!='perm' AND banned_until<=?", (now_str(),))
                con.commit()
        except Exception:
            pass
        await asyncio.sleep(300)

async def main():

    _TOUCHBET_ASYNC_LOOP = asyncio.get_running_loop()

        try:
            await asyncio.to_thread(_tb_fetch_tm_snapshot_http)
        except Exception:
            logger.exception("Initial TM HTTP snapshot failed")
    try:
        upgraded = await _tm_upgrade_pending_auto_odds()

    except Exception:
        logger.exception("Pending TM auto-odds migration failed")

        except Exception:
            logger.exception("TouchBet Mini App menu button failed")

        await dp.start_polling(bot)
